cmake_minimum_required(VERSION 3.15)
project(seal_cpu_bench_1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 默认 Release 编译；若已手动指定则保持不变
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# 更激进的优化（可按需注释）
if(NOT MSVC)
    add_compile_options(-O3 -march=native -mtune=native)
endif()

add_executable(${PROJECT_NAME} seal_cipher_compute_bench_from_user_script.cpp)
add_executable(sgf_ckks_ledger main_full.cpp)
add_executable(packing_ablation packing_ablation.cpp)
add_executable(packing_ablation_main_fast packing_ablation_main_fast.cpp)

# 优先通过 find_package(SEAL)（如果安装了 CMake 包）
find_package(SEAL QUIET)
if(SEAL_FOUND)
    if(TARGET SEAL::seal)
        target_link_libraries(${PROJECT_NAME} PRIVATE SEAL::seal)
        target_link_libraries(sgf_ckks_ledger PRIVATE SEAL::seal)
        target_link_libraries(packing_ablation PRIVATE SEAL::seal)
        target_link_libraries(packing_ablation_main_fast PRIVATE SEAL::seal)
    elseif(TARGET SEAL::seal_static)
        target_link_libraries(${PROJECT_NAME} PRIVATE SEAL::seal_static)
        target_link_libraries(sgf_ckks_ledger PRIVATE SEAL::seal_static)
        target_link_libraries(packing_ablation PRIVATE SEAL::seal_static)
        target_link_libraries(packing_ablation_main_fast PRIVATE SEAL::seal_static)
    else()
        message(WARNING "SEAL package found but target not exported; trying variables.")
        if(DEFINED SEAL_INCLUDE_DIRS)
            target_include_directories(${PROJECT_NAME} PRIVATE ${SEAL_INCLUDE_DIRS})
            target_include_directories(sgf_ckks_ledger PRIVATE ${SEAL_INCLUDE_DIRS})
            target_include_directories(packing_ablation PRIVATE ${SEAL_INCLUDE_DIRS})
            target_include_directories(packing_ablation_main_fast PRIVATE ${SEAL_INCLUDE_DIRS})
        endif()
        if(DEFINED SEAL_LIBRARIES)
            target_link_libraries(${PROJECT_NAME} PRIVATE ${SEAL_LIBRARIES})
            target_link_libraries(sgf_ckks_ledger PRIVATE ${SEAL_LIBRARIES})
            target_link_libraries(packing_ablation PRIVATE ${SEAL_LIBRARIES})
            target_link_libraries(packing_ablation_main_fast PRIVATE ${SEAL_LIBRARIES})
        endif()
    endif()
else()
    # 回退：手动指定路径。必要时在命令行传 -DSEAL_ROOT=/usr/local 或你的安装前缀
    if(NOT DEFINED SEAL_ROOT)
        set(SEAL_ROOT "/usr/local" CACHE PATH "Root of MS SEAL install (include, lib/lib64)")
    endif()
    set(SEAL_INCLUDE_DIR "${SEAL_ROOT}/include")
    find_library(SEAL_LIB NAMES seal seal-4.0 PATHS
            "${SEAL_ROOT}/lib" "${SEAL_ROOT}/lib64")
    if(NOT SEAL_LIB)
        message(FATAL_ERROR "Could not find SEAL library. Set SEAL_ROOT or install SEAL with CMake package config.")
    endif()
    target_include_directories(${PROJECT_NAME} PRIVATE "${SEAL_INCLUDE_DIR}")
    target_include_directories(sgf_ckks_ledger PRIVATE "${SEAL_INCLUDE_DIR}")
    target_include_directories(packing_ablation PRIVATE "${SEAL_INCLUDE_DIR}")
    target_include_directories(packing_ablation_main_fast PRIVATE "${SEAL_INCLUDE_DIR}")
    target_link_libraries(${PROJECT_NAME} PRIVATE "${SEAL_LIB}")
    target_link_libraries(sgf_ckks_ledger PRIVATE "${SEAL_LIB}")
    target_link_libraries(packing_ablation PRIVATE "${SEAL_LIB}")
    target_link_libraries(packing_ablation_main_fast PRIVATE "${SEAL_LIB}")
endif()

# 安装（可选）
install(TARGETS ${PROJECT_NAME} sgf_ckks_ledger packing_ablation packing_ablation_main_fast RUNTIME DESTINATION bin)
